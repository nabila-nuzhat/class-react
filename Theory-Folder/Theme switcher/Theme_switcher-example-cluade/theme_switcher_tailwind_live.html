import { createContext, useContext, useState } from 'react';

const ThemeContext = createContext(null);

export function ThemeProvider({ children }) {
  const [mode, setMode]   = useState('system');  // system | light | dark
  const [color, setColor] = useState('blue');    // blue | purple | green | rose

  // build the class string for the root element
  const themeClass = `theme-${color} mode-${mode}`;

  return (
    <ThemeContext.Provider value={{ mode, setMode, color, setColor }}>
      <div className={themeClass}>
        {children}
      </div>
    </ThemeContext.Provider>
  );
}

export const useTheme = () => useContext(ThemeContext);

import { useTheme } from './ThemeContext';

export default function Page() {
  const { mode, setMode, color, setColor } = useTheme();

  return (
    <div className="min-h-screen bg-[--bg] p-8">

      {/* mode switcher */}
      <div className="flex gap-2 mb-4">
        {['system', 'light', 'dark'].map((m) => (
          <button
            key={m}
            onClick={() => setMode(m)}
            className={`px-3 py-1 rounded text-sm border
              ${mode === m
                ? 'bg-[--btn] text-[--btntext] border-[--btn]'
                : 'bg-[--card] text-[--text] border-[--border]'
              }`}
          >
            {m}
          </button>
        ))}
      </div>

      {/* color switcher */}
      <div className="flex gap-2 mb-6">
        {['blue', 'purple', 'green', 'rose'].map((c) => (
          <button
            key={c}
            onClick={() => setColor(c)}
            className={`px-3 py-1 rounded text-sm border
              ${color === c
                ? 'bg-[--btn] text-[--btntext]'
                : 'bg-[--card] text-[--text] border-[--border]'
              }`}
          >
            {c}
          </button>
        ))}
      </div>

      {/* card using theme vars */}
      <div className="bg-[--card] border border-[--border] rounded-lg p-4">
        <h2 className="text-[--text] text-lg font-medium">Hello world</h2>
        <p className="text-[--sub] text-sm mb-3">Current theme: {color} / {mode}</p>
        <button className="bg-[--btn] text-[--btntext] px-4 py-1.5 rounded text-sm">
          Action
        </button>
      </div>

    </div>
  );
}
